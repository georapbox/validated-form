<validated-form>

A Web Component that brings automatic browser form validation to your markup, displaying accessible error messages with zero configuration or frameworks.

Source code & documentation

Demo

This demo showcases the capabilities of the <validated-form> component, demonstrating various form fields and validation scenarios. Try submitting the form with different inputs to see how it handles validation and error messages.

HTML

<validated-form>
  <form id="demo-form" autocomplete="off">
    <!-- Text / Email -->
    <div class="form-group">
      <label for="text-input">Text input (email type)</label>
      <p class="help-text" id="text-input-hint">Requires a valid email format.</p>
      <input class="form-control" type="email" id="text-input" name="text-input" required aria-describedby="text-input-hint">
      <div data-error-for="text-input" hidden></div>
    </div>

    <!-- Pattern -->
    <div class="form-group">
      <label for="pattern-input">Pattern-validated input</label>
      <p class="help-text" id="pattern-input-hint">Minimum 8 characters including letter, number, and symbol.</p>
      <input class="form-control" type="text" id="pattern-input" name="pattern-input" minlength="8" pattern="^(?=.*[A-Za-z])(?=.*\d)(?=.*[^A-Za-z\d]).{8,}$" required aria-describedby="pattern-input-hint">
      <div data-error-for="pattern-input" hidden></div>
    </div>

    <!-- Radio -->
    <fieldset class="form-group" aria-describedby="radio-hint">
      <legend>Single choice group (radio buttons)</legend>
      <p class="help-text" id="radio-hint">Exactly one option must be selected.</p>
      <label class="inline-option">
        <input type="radio" name="radio-group" value="option-a" required> Option A
      </label>
      <label class="inline-option">
        <input type="radio" name="radio-group" value="option-b" required> Option B
      </label>
      <label class="inline-option">
        <input type="radio" name="radio-group" value="option-c" required> Option C
      </label>
      <div data-error-for="radio-group" hidden></div>
    </fieldset>

    <!-- Select -->
    <div class="form-group">
      <label for="select-single">Single select</label>
      <p class="help-text" id="select-single-hint">Choose one value.</p>
      <select id="select-single" name="select-single" required aria-describedby="select-single-hint">
        <option value="">Select an option</option>
        <option value="a">Option A</option>
        <option value="b">Option B</option>
        <option value="c">Option C</option>
      </select>
      <div data-error-for="select-single" hidden></div>
    </div>

    <!-- Multi Select -->
    <div class="form-group">
      <label for="select-multiple">Multiple select</label>
      <p class="help-text" id="select-multiple-hint">Select at least one value.</p>
      <select id="select-multiple" name="select-multiple" multiple required aria-describedby="select-multiple-hint">
        <option value="a">Option A</option>
        <option value="b">Option B</option>
        <option value="c">Option C</option>
        <option value="d">Option D</option>
      </select>
      <div data-error-for="select-multiple" hidden></div>
    </div>

    <!-- File -->
    <div class="form-group">
      <label for="file-input">File upload</label>
      <p class="help-text" id="file-input-hint">Accepted formats: PDF, JPG, PNG.</p>
      <input type="file" id="file-input" name="file-input" required accept=".pdf,.jpg,.jpeg,.png" aria-describedby="file-input-hint">
      <div data-error-for="file-input" hidden></div>
    </div>

    <!-- Textarea -->
    <div class="form-group">
      <label for="textarea-input">Textarea</label>
      <p class="help-text" id="textarea-hint">Not required, but must be between 20 and 300 characters if filled.</p>
      <textarea id="textarea-input" name="textarea-input" rows="4" minlength="20" maxlength="300" aria-describedby="textarea-hint"></textarea>
      <div data-error-for="textarea-input" hidden></div>
    </div>

    <!-- Number -->
    <div class="form-group">
      <label for="number-input">Number input</label>
      <p class="help-text" id="number-input-hint">Enter a number between 5 and 10.</p>
      <input class="form-control" type="number" id="number-input" name="number-input" required min="5" max="10" aria-describedby="number-input-hint">
      <div data-error-for="number-input" hidden></div>
    </div>

    <!-- Checkbox -->
    <fieldset class="form-group" aria-describedby="checkbox-hint">
      <legend>Required checkbox</legend>
      <p class="help-text" id="checkbox-hint">Must be checked to proceed.</p>
      <label class="inline-option">
        <input type="checkbox" name="checkbox-input" required>Confirmation checkbox
      </label>
      <div data-error-for="checkbox-input" hidden></div>
    </fieldset>

    <!-- Disabled -->
    <div class="form-group">
      <label for="disabled-input">Disabled control</label>
      <p class="help-text" id="disabled-input-hint">This field is disabled. Although it is required, it will not trigger validation errors since it's not interactive.</p>
      <input id="disabled-input" name="disabled-input" type="text" disabled required aria-describedby="disabled-input-hint">
      <div data-error-for="disabled-input" hidden></div>
    </div>

    <!-- Readonly -->
    <div class="form-group">
      <label for="readonly-input">Readonly control</label>
      <p class="help-text" id="readonly-input-hint">This field is readonly. Although it is required, it will not trigger validation errors since it's not editable.</p>
      <input id="readonly-input" name="readonly-input" type="text" readonly required aria-describedby="readonly-input-hint">
      <div data-error-for="readonly-input" hidden></div>
    </div>

    <input type="hidden" name="hidden-input" value="hidden-value">

    <button type="submit">Submit</button>
  </form>
</validated-form>

JavaScript

const validatedForm = document.querySelector('validated-form');
const form = document.querySelector('form');

form.addEventListener('submit', evt => {
  evt.preventDefault();

  if (!validatedForm.isValid()) {
    return;
  }

  const formData = new FormData(form);
  const data = Object.fromEntries(formData.entries());
  console.log('Form data:', data);
});

Form submitted successfully!
